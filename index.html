// --- ゲームの初期状態パラメータ --- let player = { x: 50, y: 50, speed: 4 }; let key = { x: 700, y: 500, isCollected: false }; // 中央を避けて配置 let goal = { x: 750, y: 550 }; // ゴールを追加 let monster = { x: 400, y: 400, speed: 3.5 }; // スピードをアップ（例: 2.0 -> 3.5） // アイテムはマップ内に3個だけ配置 let items = [ { x: 100, y: 300, type: 'item1' }, { x: 300, y: 100, type: 'item2' }, { x: 600, y: 200, type: 'item3' } ]; let gameState = "PLAYING"; // "PLAYING", "GAMEOVER", "CLEAR" // --- 判定用の補助関数（円または矩形の当たり判定） --- function isColliding(a, b, radius = 20) { let dx = a.x - b.x; let dy = a.y - b.y; return (dx * dx + dy * dy) < (radius * radius); } // --- メインアップデート処理 --- function update() { if (gameState !== "PLAYING") return; // ゲームオーバー・クリア時は停止 // 1. プレイヤーの移動処理（入力に応じた更新） // movePlayer(); // 2. モンスターの追跡・移動処理 // モンスターのスピードを上げてプレイヤーへ近づける let dx = player.x - monster.x; let dy = player.y - monster.y; let angle = Math.atan2(dy, dx); monster.x += Math.cos(angle) * monster.speed; monster.y += Math.sin(angle) * monster.speed; // 【修正1】モンスターとの当たり判定（捕まったらゲームオーバー） if (isColliding(player, monster, 20)) { gameState = "GAMEOVER"; alert("モンスターに捕まった！ Game Over"); return; } // 【修正3】鍵の獲得判定 if (!key.isCollected && isColliding(player, key, 20)) { key.isCollected = true; console.log("鍵を手に入れた！ゴールに向かおう！"); } // 【修正2】ゴールの判定（鍵を持った状態で到達） if (key.isCollected && isColliding(player, goal, 25)) { gameState = "CLEAR"; alert("鍵を使って脱出成功！ CLEAR!"); return; } // アイテムの取得判定（最大3個の配列をチェック） for (let i = items.length - 1; i >= 0; i--) { if (isColliding(player, items[i], 20)) { // アイテム獲得時の効果 items.splice(i, 1); // 取得したアイテムを消去 } } }
